/* ═══════════════════════════════════════════════ PARTICLES ═══════════════════════════════════════════════ */ (function () { const cv = document.getElementById("particle-canvas"); const ctx = cv.getContext("2d"); let W, H, particles = [], accentColor = [229, 57, 53]; const RM = matchMedia("(prefers-reduced-motion: reduce)").matches; // keep the frame, drop the travel function resize() { W = cv.width = window.innerWidth; H = cv.height = window.innerHeight; } resize(); window.addEventListener("resize", resize); for ( let i = 0; i < 80; i++ // More particles ) particles.push({ x: Math.random() * 2000 - 500, y: Math.random() * 1200, vx: (Math.random() - 0.5) * 0.35, // Slightly faster vy: (Math.random() - 0.5) * 0.35, r: Math.random() * 2.5 + 0.5, // Larger: was 1.5 + 0.5 o: Math.random() * 0.5 + 0.1, // More opaque: was 0.3 + 0.05 }); function draw() { ctx.clearRect(0, 0, W, H); const [r, g, b] = accentColor; particles.forEach((p) => { if (!RM) { p.x += p.vx; p.y += p.vy; if (p.x < -50) p.x = W + 50; if (p.x > W + 50) p.x = -50; if (p.y < -50) p.y = H + 50; if (p.y > H + 50) p.y = -50; } ctx.beginPath(); ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2); ctx.fillStyle = `rgba(${r},${g},${b},${p.o})`; ctx.fill(); }); if (!RM) { for (let i = 0; i < particles.length; i++) { for (let j = i + 1; j < particles.length; j++) { const dx = particles[i].x - particles[j].x; const dy = particles[i].y - particles[j].y; const d = Math.sqrt(dx * dx + dy * dy); if (d < 140) { // Longer connection distance: was 120 ctx.beginPath(); ctx.moveTo(particles[i].x, particles[i].y); ctx.lineTo(particles[j].x, particles[j].y); ctx.strokeStyle = `rgba(${r},${g},${b},${0.5 * (1 - d / 140)})`; // More opacity: was 0.08 ctx.lineWidth = 0.8; // Thicker lines: was 0.5 ctx.stroke(); } } } } requestAnimationFrame(draw); } draw(); window.updateParticleColor = function (hex) { const r = parseInt(hex.slice(1, 3), 16); const g = parseInt(hex.slice(3, 5), 16); const b = parseInt(hex.slice(5, 7), 16); accentColor = [r, g, b]; }; })();
